cmake_minimum_required(VERSION 3.24)
project(ColorVisionSimulator VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  add_compile_options(/W4 /permissive- /utf-8 /EHsc)
  add_compile_definitions(UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
endif()
add_library(cvs_core src/core/color.cpp src/core/config.cpp)
target_include_directories(cvs_core PUBLIC src PRIVATE third_party)
include(CTest)
if(BUILD_TESTING)
  add_executable(cvs_core_tests tests/core_tests.cpp)
  target_link_libraries(cvs_core_tests PRIVATE cvs_core)
  add_test(NAME science_and_configuration COMMAND cvs_core_tests)
endif()
if(WIN32)
  if(NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "Build the Windows application with MSVC x64 and Windows SDK 10.0.22621 or later.")
  endif()
  file(READ shaders/simulation.hlsl CVS_SHADER_SOURCE)
  configure_file(cmake/shader_source.h.in generated/shader_source.h @ONLY)
  add_library(cvs_d3d src/win/d3d_renderer.cpp)
  target_include_directories(cvs_d3d PUBLIC src "${CMAKE_CURRENT_BINARY_DIR}/generated")
  target_link_libraries(cvs_d3d PUBLIC cvs_core d3d11 dxgi d3dcompiler)
  add_executable(ColorVisionSimulator WIN32 src/win/main.cpp src/win/capture.cpp src/win/app.cpp)
  target_link_libraries(ColorVisionSimulator PRIVATE cvs_d3d windowsapp dwmapi comctl32 comdlg32 shell32 shlwapi psapi)
  target_link_options(ColorVisionSimulator PRIVATE "/MANIFEST:EMBED" "/MANIFESTINPUT:${CMAKE_CURRENT_SOURCE_DIR}/src/win/app.manifest")
  if(BUILD_TESTING)
    add_executable(cvs_gpu_tests tests/gpu_tests.cpp)
    target_link_libraries(cvs_gpu_tests PRIVATE cvs_d3d)
    add_test(NAME d3d11_shader_reference COMMAND cvs_gpu_tests)
    add_executable(cvs_capture_probe tests/capture_probe.cpp src/win/capture.cpp)
    target_link_libraries(cvs_capture_probe PRIVATE cvs_d3d windowsapp)
    add_test(NAME windows_capture_probe COMMAND cvs_capture_probe)
    set_tests_properties(windows_capture_probe PROPERTIES SKIP_RETURN_CODE 77 TIMEOUT 30)
  endif()
  install(TARGETS ColorVisionSimulator RUNTIME DESTINATION .)
  install(FILES README.md LICENSE DESTINATION .)
  install(DIRECTORY LICENSES examples optional-integrations docs DESTINATION .)
endif()
